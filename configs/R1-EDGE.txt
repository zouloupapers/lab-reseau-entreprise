! ============================================================
! R1-EDGE - routeur de bordure : NAT/PAT, route par defaut, OSPF
! Modele Packet Tracer : Cisco 2911
! Remplacer CHANGE_ME par vos propres mots de passe avant de coller
! ============================================================
hostname R1-EDGE
no ip domain-lookup
service password-encryption
enable secret CHANGE_ME
username admin secret CHANGE_ME
banner motd # Acces reserve aux administrateurs du labo #
!
! --- SSH (gestion a distance) ---
ip domain-name lab.local
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
!
ip access-list standard MGMT-ACCESS
 remark Seuls les VLAN ADMIN et GESTION peuvent administrer les equipements
 permit 192.168.10.0 0.0.0.255
 permit 192.168.99.0 0.0.0.255
!
line vty 0 4
 login local
 transport input ssh
 access-class MGMT-ACCESS in
 exec-timeout 10 0
line console 0
 login local
 exec-timeout 10 0
!
! --- Interfaces ---
interface GigabitEthernet0/0
 description WAN vers ISP
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
!
interface GigabitEthernet0/1
 description Lien vers R2-CORE
 ip address 10.0.0.1 255.255.255.252
 ip nat inside
 no shutdown
!
! --- NAT/PAT : tous les VLAN utilisateurs sortent avec l'IP publique ---
ip access-list standard NAT-INSIDE
 permit 192.168.10.0 0.0.0.255
 permit 192.168.20.0 0.0.0.255
 permit 192.168.30.0 0.0.0.255
 permit 192.168.40.0 0.0.0.255
ip nat inside source list NAT-INSIDE interface GigabitEthernet0/0 overload
!
! --- Routage ---
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
router ospf 1
 router-id 1.1.1.1
 network 10.0.0.0 0.0.0.3 area 0
 default-information originate
!
end
